## 📊 Executive Summary

### Mengapa fal.ai?

fal.ai adalah **unified API gateway** yang menyediakan akses ke berbagai model AI terbaik dalam satu platform:

✅ **Multi-Provider** - Akses Kling, MiniMax, Pika, Sora, Veo, FLUX, LTX dari satu API  
✅ **Consistent Pricing** - Harga kompetitif dengan billing terpusat  
✅ **Day-0 Access** - Model terbaru tersedia di hari peluncuran  
✅ **No Vendor Lock-in** - Mudah switch antar model  
✅ **Enterprise Ready** - SOC2, scalable infrastructure

## ⚙️ Technical Implementation

### Provider Router Architecture

```typescript
// lib/fal-provider.ts

interface GenerationRequest {
    type: 'image' | 'video';
    tier: 'budget' | 'standard' | 'pro' | 'premium' | 'ultra';
    withAudio?: boolean;
    duration?: 5 | 10;
}

function selectModel(req: GenerationRequest): string {
    if (req.type === 'image') {
        switch (req.tier) {
            case 'budget': return 'fal-ai/flux/schnell';
            case 'standard': return 'fal-ai/flux-pro/v1.1';
            case 'pro': return 'fal-ai/flux-2-pro';
            case 'premium': return 'fal-ai/flux-2-pro';
            case 'ultra': return 'fal-ai/gpt-image-1';
        }
    }
    
    if (req.type === 'video') {
        if (req.withAudio) {
            switch (req.tier) {
                case 'budget': 
                case 'standard': 
                case 'pro': return 'fal-ai/kling-video/v2.6/pro';  // Lip sync
                case 'premium': return 'fal-ai/veo/v3.1';          // Cinematic + audio
                case 'ultra': return 'fal-ai/sora-2/text-to-video'; // Best overall
            }
        }
        
        switch (req.tier) {
            case 'budget': return 'fal-ai/ltxv-2/fast';
            case 'standard': return 'fal-ai/minimax/hailuo-2.3-fast';
            case 'pro': return 'fal-ai/kling-video/v2.5/pro';
            case 'premium': return 'fal-ai/veo/v3.0';              // Cinematic
            case 'ultra': return 'fal-ai/sora-2/text-to-video';
        }
    }
    
    return 'fal-ai/flux/schnell'; // default
}

function calculateTokens(model: string, duration?: number): number {
    const tokenMap: Record<string, number> = {
        // Images
        'fal-ai/flux/schnell': 2,
        'fal-ai/flux-pro/v1.1': 8,
        'fal-ai/flux-2-pro': 14,
        'fal-ai/gpt-image-1': 20,
        
        // Videos (base for 5s)
        'fal-ai/ltxv-2/fast': 12,
        'fal-ai/minimax/hailuo-2.3-fast': 20,
        'fal-ai/kling-video/v2.5/pro': 48,
        'fal-ai/kling-video/v2.6/pro': 72,  // with audio + lip sync
        'fal-ai/pika/v2.2': 60,
        'fal-ai/veo/v3.0': 72,              // cinematic, no audio
        'fal-ai/veo/v3.1': 100,             // cinematic + native audio
        'fal-ai/sora-2/text-to-video': 160,
    };
    
    let tokens = tokenMap[model] || 10;
    
    // Double for 10s videos
    if (duration === 10) {
        tokens = Math.floor(tokens * 1.7);
    }
    
    return tokens;
}
```

### API Integration Example

```typescript
// lib/fal.ts
import * as fal from "@fal-ai/serverless-client";

fal.config({
    credentials: process.env.FAL_KEY,
});

export async function generateImage(prompt: string, tier: string) {
    const model = selectModel({ type: 'image', tier });
    
    const result = await fal.subscribe(model, {
        input: {
            prompt,
            image_size: tier === 'ultra' ? '4k' : 'landscape_16_9',
            num_images: 1,
        },
    });
    
    return result.images[0].url;
}

export async function generateVideo(
    prompt: string, 
    tier: string,
    withAudio: boolean = false,
    duration: 5 | 10 = 5
) {
    const model = selectModel({ type: 'video', tier, withAudio, duration });
    
    const result = await fal.subscribe(model, {
        input: {
            prompt,
            duration: `${duration}`,
            aspect_ratio: '16:9',
        },
    });
    
    return result.video.url;
}
```
